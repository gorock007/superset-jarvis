      +// ErrRateLimited is returned when a client exceeds its window.
      +var ErrRateLimited = errors.New("rate limit exceeded")
      +
      +func (p *RateLimit) Handle(ctx context.Context, req *Request) (*Response, error) {
      +	key := "rl:" + req.ClientID
      +	n, err := p.rdb.Incr(ctx, key).Result()
      +	if err != nil {
      +		return nil, fmt.Errorf("ratelimit: incr: %w", err)
      +	}
      +	if n == 1 {
      +		p.rdb.Expire(ctx, key, p.window)
      +	}
      +	if n > p.max {
      +		ttl, _ := p.rdb.TTL(ctx, key).Result()
      +		return &Response{
      +			Status: 429,
      +			Body:   fmt.Sprintf("Rate limit exceeded, try again in %ds", int(ttl.Seconds())),
      +		}, ErrRateLimited
      +	}
      +	return p.next.Handle(ctx, req)
      +}
